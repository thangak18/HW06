```bash
curl -i -X POST 'http://localhost:3000/api/register' \
     -H 'Content-Type: application/json' \
     -H 'X-Student-Id: 23127060' \
     -d '{"name": "Victim", "email": "api.b12.victim.23127060@test.local", "password": "Api1234!"}'
```
